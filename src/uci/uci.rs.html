33#[allow(dead_code)]
34#[derive(Clone)]
35enum Value {
36    Check(bool),
37    Spin(i64),
38    String(String),
39    Button,
40    Combo(String),
41}
42
43impl Value {
44    fn check(&self) -> Option<bool> {
45        if let Value::Check(value) = self {
46            Some(*value)
47        } else {
48            None
49        }
50    }
51
52    fn spin(&self) -> Option<i64> {
53        if let Value::Spin(value) = self {
54            Some(*value)
55        } else {
56            None
57        }
58    }
59
60    fn string(&self) -> Option<String> {
61        if let Value::String(value) = self {
62            Some(value.clone())
63        } else {
64            None
65        }
66    }
67
68    fn combo(&self) -> Option<String> {
69        if let Value::Combo(value) = self {
70            Some(value.clone())
71        } else {
72            None
73        }
74    }
75}

77#[allow(dead_code)]
78#[derive(Clone)]
79enum Kind {
80    Check,
81    Spin { min: Option<i64>, max: Option<i64> },
82    Combo(Vec<String>),
83    Button,
84    String,
85}
86
87type HookFn = fn(&Option<Value>) -> Option<GuiCmd>;
88
89#[derive(Clone)]
90struct Opt {
91    pub name: &'static str,
92    pub kind: Kind,
93    pub value: Value,
94    pub default: Value,
95    pub hook: Option<HookFn>,
96}
97
98impl Opt {
99    pub fn new(name: &'static str, kind: Kind, value: Value, hook: Option<HookFn>) -> Self {
100        Self {
101            name,
102            kind,
103            value: value.clone(),
104            default: value,
105            hook,
106        }
107    }
108}
109
110struct Registry {
111    opts: Vec<Opt>,
112    extra: HashMap<String, Option<String>>,
113}
114
115impl Registry {
116    pub fn new() -> Self {
117        Self {
118            opts: Vec::new(),
119            extra: HashMap::default(),
120        }
121    }
122
123    pub fn define(&mut self, name: &'static str, kind: Kind, value: Value, hook: Option<HookFn>) {
124        self.opts.push(Opt::new(name, kind, value, hook));
125    }
126
127    fn option_definitions(&self) -> Vec<OptionType> {
128        self.opts
129            .iter()
130            .map(|opt| match opt.kind {
131                Kind::Check => OptionType::Check {
132                    name: opt.name.to_string(),
133                    default: opt.default.check(),
134                },
135                Kind::Spin { min, max } => OptionType::Spin {
136                    name: opt.name.to_string(),
137                    default: opt.default.spin().unwrap(),
138                    min,
139                    max,
140                },
141                Kind::Combo(ref vars) => OptionType::Combo {
142                    name: opt.name.to_string(),
143                    default: opt.default.combo().unwrap(),
144                    vars: vars.clone(),
145                },
146                Kind::Button => OptionType::Button(opt.name.to_string()),
147                Kind::String => OptionType::String {
148                    name: opt.name.to_string(),
149                    default: Some(opt.default.string().unwrap_or("<empty>".to_string())),
150                },
151            })
152            .collect()
153    }
154
155    pub fn set(&mut self, name: &String, value: &Option<String>) -> Option<GuiCmd> {
156        for opt in self.opts.iter_mut() {
157            if opt.name == name {
158                match opt.kind {
159                    Kind::Check => {
160                        if let Some(value) = value {
161                            opt.value = Value::Check(value.to_lowercase() == "true");
162                        } else {
163                            error!("No value provided for option of type Check {name}")
164                        }
165                    }
166                    Kind::Spin { min, max } => {
167                        if let Some(value) = value {
168                            let v = value.parse::<i64>();
169                            if let Ok(v) = v {
170                                let min = min.unwrap_or(i64::MIN);
171                                let max = max.unwrap_or(i64::MAX);
172                                if v > min && v < max {
173                                    opt.value = Value::Spin(v);
174                                } else {
175                                    error!(
176                                        "Value of type Spin is out of bounds. {name} must be in [{min}, {max}]. Attempting to set it to {v}"
177                                    )
178                                }
179                            } else {
180                                error!(
181                                    "Value of type Spin cannot be interpreted as an integer {name}"
182                                );
183                            }
184                        } else {
185                            error!("No value provided for option of type Check {name}")
186                        }
187                    }
188                    Kind::String => {
189                        if let Some(value) = value {
190                            opt.value = Value::String(value.clone());
191                        } else {
192                            error!("No value provided for option of type String {name}")
193                        }
194                    }
195                    Kind::Button => {
196                        unimplemented!();
197                    }
198                    Kind::Combo(_) => {
199                        if let Some(value) = value {
200                            opt.value = Value::Combo(value.clone());
201                        } else {
202                            error!("No value provided for option of type String {name}")
203                        }
204                    }
205                }
206                return opt.hook.and_then(|f| f(&Some(opt.value.clone())));
207            }
208        }
209
210        self.extra.insert(name.to_string(), value.clone());
211        None
212    }
213}

215fn main() -> Result<()> {
216    env_logger::init();
217
218    let (gui_tx, engine_rx): (Sender<GuiCmd>, Receiver<GuiCmd>) = mpsc::channel();
219    let (engine_tx, gui_rx): (Sender<EngineCmd>, Receiver<EngineCmd>) = mpsc::channel();
220
221    let mut registry = Registry::new();
222    registry.define(
223        "OpeningBookFile",
224        Kind::String,
225        Value::String("".to_string()),
226        Some(|value| {
227            Some(GuiCmd::OpeningBook(
228                value.clone().unwrap().string().unwrap(),
229            ))
230        }),
231    );
232
233    let t = thread::spawn(move || {
234        loop {
235            let mut buffer = String::new();
236            io::stdin().read_line(&mut buffer).unwrap();
237            let buffer = buffer.replace('\n', "");
238
239            let commands = match dbg!(UciGuiCmd::parse(dbg!(&buffer)).unwrap()) {
240                UciGuiCmd::Uci => {
241                    let mut cmds = vec![
242                        UciEngineCmd::IdName("FChess".to_string()),
243                        UciEngineCmd::IdAuthor("João Freitas".to_string()),
244                    ];
245
246                    for option in registry.option_definitions() {
247                        cmds.push(UciEngineCmd::Option(option));
248                    }
249
250                    cmds.push(UciEngineCmd::UciOk);
251                    cmds
252                }
253                UciGuiCmd::UciNewGame => vec![],
254                UciGuiCmd::IsReady => vec![UciEngineCmd::ReadyOk],
255                UciGuiCmd::Quit => std::process::exit(0),
256                UciGuiCmd::Stop => vec![],
257                UciGuiCmd::Position { pos, moves } => {
258                    gui_tx.send(GuiCmd::Position { pos, moves }).unwrap();
259                    vec![]
260                }
261                UciGuiCmd::Go(_) => {
262                    gui_tx.send(GuiCmd::Go).unwrap();
263                    if let EngineCmd::BestMove(mov) = gui_rx.recv().unwrap() {
264                        vec![UciEngineCmd::BestMove { mov, ponder: None }]
265                    } else {
266                        panic!()
267                    }
268                }
269                UciGuiCmd::SetOption { name, value } => {
270                    let cmd = registry.set(&name, &value);
271                    if let Some(cmd) = cmd {
272                        gui_tx.send(cmd).unwrap();
273                    }
274                    vec![]
275                }
276                UciGuiCmd::Fen => {
277                    gui_tx.send(GuiCmd::Fen).unwrap();
278                    let cmd = gui_rx.recv().unwrap();
279                    if let EngineCmd::Fen(fen) = cmd {
280                        vec![UciEngineCmd::Info(
281                            InfoSubCmdBuilder::new().with_string(&fen).build(),
282                        )]
283                    } else {
284                        panic!()
285                    }
286                }
287                _ => panic!(),
288            };
289
290            for command in commands {
291                to_client(dbg!(&command.write()));
292            }
293        }
294    });
295
296    let _engine_thread = thread::spawn(move || {
297        let mut board =
298            Board::from_fen("rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR b KQkq e3 0 1").unwrap();
299        let mut solver = Solver::new();
300        loop {
301            let cmd = engine_rx.recv().unwrap();
302            match cmd {
303                GuiCmd::Fen => engine_tx
304                    .send(EngineCmd::Fen(board.to_fen().unwrap()))
305                    .unwrap(),
306                GuiCmd::Position { pos, moves } => {
307                    match pos {
308                        PositionSubCmd::StartPos => {
309                            board = Board::from_basic_board();
310                        }
311                        PositionSubCmd::Fen(fen) => {
312                            board = Board::from_fen(&fen).unwrap();
313                        }
314                    };
315
316                    for mov in moves {
317                        board = board.apply(&mov).unwrap();
318                    }
319                }
320                GuiCmd::Go => {
321                    engine_tx
322                        .send(EngineCmd::BestMove(solver.best_move(&board).unwrap()))
323                        .unwrap();
324                }
325                GuiCmd::OpeningBook(filename) => {
326                    solver.set_opening_book(Book::from_filename(&filename));
327                }
328            };
329        }
330    });
331
332    t.join().unwrap();
333    _engine_thread.join().unwrap();
334    Ok(())
335}
