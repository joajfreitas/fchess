15fn to_client(file: &mut File, output: &str) {
16    file.write_all(format!("engine - {}\n", output).as_bytes())
17        .unwrap();
18    println!("{}", output);
19}

21enum GuiCmd {
22    Fen,
23    Position {
24        pos: PositionSubCmd,
25        moves: Vec<Move>,
26    },
27    Go,
28}
29
30enum EngineCmd {
31    Fen(String),
32    BestMove(Move),
33}
34
35fn main() -> Result<()> {
36    let (gui_tx, engine_rx): (Sender<GuiCmd>, Receiver<GuiCmd>) = mpsc::channel();
37    let (engine_tx, gui_rx): (Sender<EngineCmd>, Receiver<EngineCmd>) = mpsc::channel();
38
39    let t = thread::spawn(move || {
40        let mut file = File::create("/home/joaj/sources/fchess/log1.txt").unwrap();
41        loop {
42            let mut buffer = String::new();
43            io::stdin().read_line(&mut buffer).unwrap();
44            let buffer = buffer.replace('\n', "");
45
46            let commands = match dbg!(UciGuiCmd::parse(dbg!(&buffer))).unwrap() {
47                UciGuiCmd::Uci => vec![
48                    UciEngineCmd::IdName("FChess".to_string()),
49                    UciEngineCmd::IdAuthor("João Freitas".to_string()),
50                    UciEngineCmd::Option(OptionTypes::String {
51                        name: "Opening book".to_string(),
52                        default: None,
53                    }),
54                    UciEngineCmd::UciOk,
55                ],
56                UciGuiCmd::UciNewGame => vec![],
57                UciGuiCmd::IsReady => vec![UciEngineCmd::ReadyOk],
58                UciGuiCmd::Quit => std::process::exit(0),
59                UciGuiCmd::Stop => vec![],
60                UciGuiCmd::Position { pos, moves } => {
61                    gui_tx.send(GuiCmd::Position { pos, moves }).unwrap();
62                    vec![]
63                }
64                UciGuiCmd::Go(_) => {
65                    gui_tx.send(GuiCmd::Go).unwrap();
66                    if let EngineCmd::BestMove(mov) = gui_rx.recv().unwrap() {
67                        vec![UciEngineCmd::BestMove {
68                            mov,
69                            ponder: None,
70                        }]
71                    } else {
72                        panic!()
73                    }
74                }
75                UciGuiCmd::SetOption { name: _, value: _ } => unimplemented!(),
76                UciGuiCmd::Fen => {
77                    gui_tx.send(GuiCmd::Fen).unwrap();
78                    let cmd = gui_rx.recv().unwrap();
79                    if let EngineCmd::Fen(fen) = cmd {
80                        vec![UciEngineCmd::Info(
81                            InfoSubCmdBuilder::new().with_string(&fen).build(),
82                        )]
83                    } else {
84                        panic!()
85                    }
86                }
87                _ => panic!(),
88            };
89
90            for command in commands {
91                to_client(&mut file, &write_engine_command(&command));
92            }
93
94            file.write_all(format!("client - {buffer}\n").as_bytes())
95                .unwrap();
96        }
97    });
98
99    let _engine_thread = thread::spawn(move || {
100        let mut board =
101            Board::from_fen("rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR b KQkq e3 0 1").unwrap();
102        let mut solver = Solver::new();
103        loop {
104            let cmd = engine_rx.recv().unwrap();
105            match cmd {
106                GuiCmd::Fen => engine_tx
107                    .send(EngineCmd::Fen(board.to_fen().unwrap()))
108                    .unwrap(),
109                GuiCmd::Position { pos, moves } => {
110                    match pos {
111                        PositionSubCmd::StartPos => {
112                            board = Board::from_basic_board();
113                        }
114                        PositionSubCmd::Fen(fen) => {
115                            board = Board::from_fen(&fen).unwrap();
116                        }
117                    };
118
119                    for mov in moves {
120                        board = board.apply(&mov).unwrap();
121                    }
122                }
123                GuiCmd::Go => {
124                    engine_tx
125                        .send(EngineCmd::BestMove(solver.best_move(&board).unwrap()))
126                        .unwrap();
127                }
128            };
129        }
130    });
131
132    t.join().unwrap();
133    _engine_thread.join().unwrap();
134    Ok(())
135}
