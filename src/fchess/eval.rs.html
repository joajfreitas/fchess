72    fn piece_value(&self, piece_type: &ColoredPieceType, side: &Side) -> i32 {
73        self.piece_value[*piece_type as usize] * if *side == Side::White { 1 } else { -1 }
74    }
75
76    fn piece_square_table(&self, index: usize, piece_type: &ColoredPieceType, side: &Side) -> i32 {
77        let index = if piece_type.is_black() {
78            index
79        } else {
80            63 - index
81        };
82        let s = match (*piece_type).try_into().unwrap() {
83            PieceType::Pawn => self.pawn_table[index],
84            PieceType::Rook => self.rook_table[index],
85            PieceType::King => self.king_middle_table[index],
86            PieceType::Queen => self.queen_table[index],
87            PieceType::Bishop => self.bishop_table[index],
88            PieceType::Knight => self.knight_table[index],
89        };
90        s * (if piece_type.side() == *side { 1 } else { -1 })
91    }
92
93    pub fn eval(&self, board: &Board, side: &Side) -> i32 {
94        let mut s: i32 = 0;
95        for piece_type in COLORED_PIECE_TYPES {
96            for i in BitIter::from(board.get_piece_mask(piece_type)) {
97                s += self.piece_value(&piece_type, side)
98                    + self.piece_square_table(i, &piece_type, side);
99            }
100        }
101
102        s
103    }
104}

132        let basic_board_builder = BoardBuilder::new().with_basic_board();
133
134        let e2e4 = basic_board_builder.with_move(&mv!("e2e4")?)?.build();

136
137        let eval = Eval::new();

139        let h2h4_eval = eval.eval(&h2h4, &Side::White);
