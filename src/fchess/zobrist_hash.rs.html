794    fn typing_cast(piece: &Piece) -> u32 {
795        match piece.get_type() {
796            ColoredPieceType::BlackPawn => 0,
797            ColoredPieceType::WhitePawn => 1,
798            ColoredPieceType::BlackKnight => 2,
799            ColoredPieceType::WhiteKnight => 3,
800            ColoredPieceType::BlackBishop => 4,
801            ColoredPieceType::WhiteBishop => 5,
802            ColoredPieceType::BlackRook => 6,
803            ColoredPieceType::WhiteRook => 7,
804            ColoredPieceType::BlackQueen => 8,
805            ColoredPieceType::WhiteQueen => 9,
806            ColoredPieceType::BlackKing => 10,
807            ColoredPieceType::WhiteKing => 11,
808            _ => panic!(),
809        }
810    }
811
812    let mut piece_hash: u64 = 0;
813    for piece in board {
814        let zobrist_table_index = 64 * typing_cast(&piece) + piece.get_square().get_index() as u32;
815        piece_hash ^= ZOBRIST_TABLE[zobrist_table_index as usize];
816    }
817
818    let mut castle_hash = 0;
819
820    if board.get_castling_white_short() {
821        castle_hash ^= ZOBRIST_TABLE[768];
822    }
823    if board.get_castling_white_long() {
824        castle_hash ^= ZOBRIST_TABLE[768 + 1];
825    }
826    if board.get_castling_black_short() {
827        castle_hash ^= ZOBRIST_TABLE[768 + 2];
828    }
829    if board.get_castling_black_long() {
830        castle_hash ^= ZOBRIST_TABLE[768 + 3];
831    }
832
833    let enpassant_hash = board.get_enpassant().map_or(0, |square| {
834        let mut ep_mask = square.get_mask();
835        ep_mask = if board.get_turn() == Side::White {
836            ep_mask.shift_down()
837        } else {
838            ep_mask.shift_up()
839        };
840        ep_mask = ep_mask.shift_left() | ep_mask.shift_right();
841
842        if ep_mask & board.get_piece_mask(PieceType::Pawn.with_color(board.get_turn())) > 0 {
843            ZOBRIST_TABLE[772 + square.get_file() as usize]
844        } else {
845            0
846        }
847    });
848
849    let side_hash = match board.get_turn() {
850        Side::White => ZOBRIST_TABLE[780],
851        Side::Black => 0,
852    };
853
854    piece_hash ^ castle_hash ^ side_hash ^ enpassant_hash
