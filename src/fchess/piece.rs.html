7const LAST_WHITE_OFFSET: u32 = 6;
8const LAST_BLACK_OFFSET: u32 = 12;

10#[allow(dead_code)]
11#[derive(FromPrimitive, ToPrimitive)]
12pub enum PieceType {
13    Pawn = 0,
14    Rook = 1,
15    Knight = 2,
16    Bishop = 3,
17    Queen = 4,
18    King = 5,
19}

21impl PieceType {
22    pub fn with_color(self, side: Side) -> ColoredPieceType {
23        let side_offset: u32 = match side {
24            Side::White => 0,
25            Side::Black => LAST_WHITE_OFFSET,
26        };
27
28        num::FromPrimitive::from_u32(num::ToPrimitive::to_u32(&self).unwrap() + side_offset)
29            .unwrap()
30    }
31}

52impl fmt::Display for ColoredPieceType {
53    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
54        let pieces = [
55            "♟︎", "♜", "♞", "♝", "♛", "♚", "♙", "♖", "♘", "♗", "♕", "♔", "*", "+", " ",
56        ];
57        f.write_str(pieces[*self as usize])
58    }

61impl ColoredPieceType {
62    pub fn is_black(self: &ColoredPieceType) -> bool {
63        let offset = num::ToPrimitive::to_u32(self).unwrap();
64
65        (LAST_WHITE_OFFSET..LAST_BLACK_OFFSET).contains(&offset)
66    }
67
68    pub fn is_white(self: &ColoredPieceType) -> bool {
69        let offset = num::ToPrimitive::to_u32(self).unwrap();
70
71        offset < LAST_WHITE_OFFSET
72    }
73
74    pub fn from_char(c: &char) -> Option<ColoredPieceType> {
75        match c {
76            'P' => Some(ColoredPieceType::WhitePawn),
77            'R' => Some(ColoredPieceType::WhiteRook),
78            'N' => Some(ColoredPieceType::WhiteKnight),
79            'B' => Some(ColoredPieceType::WhiteBishop),
80            'Q' => Some(ColoredPieceType::WhiteQueen),
81            'K' => Some(ColoredPieceType::WhiteKing),
82            'p' => Some(ColoredPieceType::BlackPawn),
83            'r' => Some(ColoredPieceType::BlackRook),
84            'n' => Some(ColoredPieceType::BlackKnight),
85            'b' => Some(ColoredPieceType::BlackBishop),
86            'q' => Some(ColoredPieceType::BlackQueen),
87            'k' => Some(ColoredPieceType::BlackKing),
88            _ => None,
89        }
90    }
