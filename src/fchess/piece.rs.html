21impl PieceType {
22    pub fn with_color(self, side: Side) -> ColoredPieceType {
23        let side_offset: u32 = match side {
24            Side::White => 0,
25            Side::Black => LAST_WHITE_OFFSET,
26        };
27
28        num::FromPrimitive::from_u32(num::ToPrimitive::to_u32(&self).unwrap() + side_offset)
29            .unwrap()
30    }
31}

33impl TryFrom<ColoredPieceType> for PieceType {

40            ColoredPieceType::WhiteKnight => Ok(Self::Knight),

43            ColoredPieceType::WhiteKing => Ok(Self::King),

85impl ColoredPieceType {
86    pub fn is_black(&self) -> bool {
87        let offset = num::ToPrimitive::to_u32(self).unwrap();
88
89        (LAST_WHITE_OFFSET..LAST_BLACK_OFFSET).contains(&offset)
90    }
91
92    pub fn is_white(&self) -> bool {
93        let offset = num::ToPrimitive::to_u32(self).unwrap();
94
95        offset < LAST_WHITE_OFFSET
96    }
97
98    pub fn side(&self) -> Side {
99        if self.is_white() {
100            Side::White
101        } else {
102            Side::Black
103        }
104    }
105

115            'r' => Some(ColoredPieceType::BlackRook),
